# cu126, NOT cu130: driver 566.07 on the dev host cannot run a CUDA 13 build,
# and the CUDA toolkit used to compile mini-flash-attention must match this
# major version or torch's CUDAExtension raises outright. See PROGRESS.md.
# Reinstalling over an existing cu130 needs --force-reinstall: pip treats
# 2.9.1+cu130 as already satisfying a bare torch==2.9.1 and silently no-ops.
torch==2.9.1+cu126

# Official triton publishes no Windows wheels; triton-windows is the substitute.
# Keep both on 3.5.x — that is what torch 2.9 pairs with, and a newer triton
# risks breaking the @torch.compile on MLP.forward.
triton==3.5.1; sys_platform != "win32"
triton-windows==3.5.1.post24; sys_platform == "win32"

# Built from source (CUTLASS / C++20 CUDA). This line works as-is on Linux.
# On Windows it does NOT: upstream setup.py uses GCC flag spellings and a
# lib64 path that do not exist there, and the CUTLASS submodule overruns
# MAX_PATH from a deep directory. PROGRESS.md -> Environment State has the
# patched build recipe; install it separately there.
mini_flash_attention @ git+https://github.com/w4096/mini-flash-attention.git ; sys_platform != "win32"
